package com.example

import android.app.Application
import coil.ImageLoader
import coil.ImageLoaderFactory
import coil.decode.SvgDecoder
import coil.disk.DiskCache
import coil.memory.MemoryCache
import com.example.data.local.AppDatabase
import com.example.data.remote.RadioBrowserApiService
import com.example.data.repository.RadioRepository
import com.example.player.RadioPlayerManager
import okhttp3.OkHttpClient
import java.util.concurrent.TimeUnit

class AutoRadioApplication : Application(), ImageLoaderFactory {

    lateinit var database: AppDatabase
        private set

    lateinit var repository: RadioRepository
        private set

    lateinit var playerManager: RadioPlayerManager
        private set

    lateinit var weatherApiService: com.example.data.remote.OpenMeteoApiService
        private set

    override fun onCreate() {
        super.onCreate()
        database = AppDatabase.getDatabase(this)
        repository = RadioRepository(
            apiService = RadioBrowserApiService.create(),
            favoriteDao = database.favoriteStationDao()
        )
        playerManager = RadioPlayerManager.getInstance(this)
        weatherApiService = com.example.data.remote.OpenMeteoApiService.create()
    }

    override fun newImageLoader(): ImageLoader {
        val okHttpClient = OkHttpClient.Builder()
            .connectTimeout(15, TimeUnit.SECONDS)
            .readTimeout(15, TimeUnit.SECONDS)
            .followRedirects(true)
            .followSslRedirects(true)
            .addInterceptor { chain ->
                val request = chain.request().newBuilder()
                    .header("User-Agent", "Mozilla/5.0 (Linux; Android 14) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0 Mobile Safari/537.36 AutoRadio/1.0")
                    .build()
                chain.proceed(request)
            }
            .build()

        return ImageLoader.Builder(this)
            .okHttpClient(okHttpClient)
            .components {
                add(SvgDecoder.Factory())
            }
            .memoryCache {
                MemoryCache.Builder(this)
                    .maxSizePercent(0.25)
                    .build()
            }
            .diskCache {
                DiskCache.Builder()
                    .directory(cacheDir.resolve("image_cache"))
                    .maxSizeBytes(50L * 1024 * 1024)
                    .build()
            }
            .respectCacheHeaders(false)
            .crossfade(true)
            .build()
    }
}

package com.example.ui

import android.app.Application

class MainViewModel(application: Application) : AndroidViewModel(application) {

    private val app = application as AutoRadioApplication
    private val repository = app.repository
    private val playerManager = app.playerManager

    val currentStation: StateFlow<RadioStation?> = playerManager.currentStation
    val playbackStatus: StateFlow<PlaybackStatus> = playerManager.playbackStatus
    val rdsInfo: StateFlow<RdsInfo> = playerManager.rdsInfo
    val errorMessage: StateFlow<String?> = playerManager.errorMessage
    val sleepTimerRemaining: StateFlow<Int?> = playerManager.sleepTimerSeconds
    val isMuted: StateFlow<Boolean> = playerManager.isMuted

    val favoriteStations: StateFlow<List<RadioStation>> = repository.favoriteStations
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    val recentStations: StateFlow<List<RadioStation>> = repository.recentStations
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    private val _topStations = MutableStateFlow<List<RadioStation>>(emptyList())
    val topStations: StateFlow<List<RadioStation>> = _topStations.asStateFlow()

    private val _dutchStations = MutableStateFlow<List<RadioStation>>(emptyList())
    val dutchStations: StateFlow<List<RadioStation>> = _dutchStations.asStateFlow()

    private val _danceStations = MutableStateFlow<List<RadioStation>>(emptyList())
    val danceStations: StateFlow<List<RadioStation>> = _danceStations.asStateFlow()

    private val _rockStations = MutableStateFlow<List<RadioStation>>(emptyList())
    val rockStations: StateFlow<List<RadioStation>> = _rockStations.asStateFlow()

    private val _newsStations = MutableStateFlow<List<RadioStation>>(emptyList())
    val newsStations: StateFlow<List<RadioStation>> = _newsStations.asStateFlow()

    private val _searchResults = MutableStateFlow<List<RadioStation>>(emptyList())
    val searchResults: StateFlow<List<RadioStation>> = _searchResults.asStateFlow()

    private val _searchQuery = MutableStateFlow("")
    val searchQuery: StateFlow<String> = _searchQuery.asStateFlow()

    private val _selectedCountryFilter = MutableStateFlow<String?>(null)
    val selectedCountryFilter: StateFlow<String?> = _selectedCountryFilter.asStateFlow()

    private val _selectedGenreFilter = MutableStateFlow<String?>(null)
    val selectedGenreFilter: StateFlow<String?> = _selectedGenreFilter.asStateFlow()

    private val _isSearching = MutableStateFlow(false)
    val isSearching: StateFlow<Boolean> = _isSearching.asStateFlow()

    private val _isLoadingHome = MutableStateFlow(false)
    val isLoadingHome: StateFlow<Boolean> = _isLoadingHome.asStateFlow()

    private val _isCarMode = MutableStateFlow(false)
    val isCarMode: StateFlow<Boolean> = _isCarMode.asStateFlow()

    private val _showFullPlayer = MutableStateFlow(false)
    val showFullPlayer: StateFlow<Boolean> = _showFullPlayer.asStateFlow()

    private val _weatherInfo = MutableStateFlow<com.example.model.WeatherInfo?>(null)
    val weatherInfo: StateFlow<com.example.model.WeatherInfo?> = _weatherInfo.asStateFlow()

    private var searchJob: Job? = null

    init {
        loadHomeData()
        loadWeatherData()
    }

    fun loadWeatherData() {
        viewModelScope.launch {
            try {
                val resp = app.weatherApiService.getCurrentWeather()
                val current = resp.current
                if (current != null) {
                    val info = com.example.model.WeatherInfo.fromWmoCode(
                        code = current.weatherCode,
                        temp = current.temperature,
                        isDayTime = current.isDay != 0
                    )
                    _weatherInfo.value = info
                }
            } catch (_: Exception) {
                // Default fallback nice weather
                _weatherInfo.value = com.example.model.WeatherInfo(
                    temperatureCelsius = 19,
                    condition = com.example.model.WeatherCondition.PARTLY_CLOUDY,
                    conditionDescription = "Licht bewolkt",
                    isDay = true
                )
            }
        }
    }

    fun loadHomeData() {
        viewModelScope.launch {
            _isLoadingHome.value = true
            try {
                _dutchStations.value = repository.getDutchStations()
                _topStations.value = repository.getTopStations()
                _danceStations.value = repository.getStationsByGenre("dance")
                _rockStations.value = repository.getStationsByGenre("rock")
                _newsStations.value = repository.getStationsByGenre("news")
            } catch (_: Exception) {
            } finally {
                _isLoadingHome.value = false
            }
        }
    }

    fun playStation(station: RadioStation, contextPlaylist: List<RadioStation>? = null) {
        viewModelScope.launch {
            repository.recordPlayed(station)
            val isFav = repository.isFavorite(station.stationUuid)
            val updated = station.copy(isFavorite = isFav)
            val playlist = contextPlaylist ?: listOf(updated)
            playerManager.playStation(updated, playlist)
        }
    }

    fun togglePlayPause() {
        playerManager.togglePlayPause()
    }

    fun nextStation() {
        playerManager.nextStation()
    }

    fun previousStation() {
        playerManager.previousStation()
    }

    fun toggleFavorite(station: RadioStation) {
        viewModelScope.launch {
            val isNowFav = repository.toggleFavorite(station)
            
            // Sync current station if playing
            val current = currentStation.value
            if (current != null && current.stationUuid == station.stationUuid) {
                // Trigger updated currentStation inside playerManager
                playerManager.updateCurrentStationFavorite(isNowFav)
            }
            
            // Sync all in-memory lists
            updateListsFavoriteState(station.stationUuid, isNowFav)
        }
    }

    private fun updateListsFavoriteState(uuid: String, isFav: Boolean) {
        fun update(list: List<RadioStation>): List<RadioStation> {
            return list.map { if (it.stationUuid == uuid) it.copy(isFavorite = isFav) else it }
        }
        _topStations.value = update(_topStations.value)
        _dutchStations.value = update(_dutchStations.value)
        _danceStations.value = update(_danceStations.value)
        _rockStations.value = update(_rockStations.value)
        _newsStations.value = update(_newsStations.value)
        _searchResults.value = update(_searchResults.value)
    }

    fun setCarMode(enabled: Boolean) {
        _isCarMode.value = enabled
    }

    fun setShowFullPlayer(show: Boolean) {
        _showFullPlayer.value = show
    }

    fun setSleepTimer(minutes: Int?) {
        playerManager.setSleepTimer(minutes)
    }

    fun toggleMute() {
        playerManager.toggleMute()
    }

    fun onSearchQueryChanged(query: String) {
        _searchQuery.value = query
        triggerDebouncedSearch()
    }

    fun selectCountryFilter(countryCode: String?) {
        _selectedCountryFilter.value = countryCode
        triggerDebouncedSearch()
    }

    fun selectGenreFilter(genre: String?) {
        _selectedGenreFilter.value = if (_selectedGenreFilter.value == genre) null else genre
        triggerDebouncedSearch()
    }

    private fun triggerDebouncedSearch() {
        searchJob?.cancel()
        searchJob = viewModelScope.launch {
            delay(350)
            val query = _searchQuery.value.trim()
            val country = _selectedCountryFilter.value
            val genre = _selectedGenreFilter.value

            if (query.isEmpty() && country == null && genre == null) {
                _searchResults.value = emptyList()
                return@launch
            }

            _isSearching.value = true
            try {
                val results = repository.searchStations(
                    query = query,
                    countryCode = country,
                    genre = genre
                )
                _searchResults.value = results
            } catch (_: Exception) {
                _searchResults.value = emptyList()
            } finally {
                _isSearching.value = false
            }
        }
    }
}

@Composable
fun StationLogoImage(
    imageUrl: String?,
    stationName: String,
    modifier: Modifier = Modifier,
    size: Dp = 56.dp,
    cornerRadius: Dp = 10.dp
) {
    // Sanitize image URL if present: convert HTTP images to secure HTTPS or fix common radio-browser issues
    val sanitizedUrl = imageUrl?.trim()?.takeIf { it.isNotBlank() }?.let { url ->
        when {
            // Wikipedia thumbnail URLs frequently return HTTP 400 when missing correct parameters
            url.contains("upload.wikimedia.org") && url.contains("/thumb/") -> null
            url.startsWith("http://") -> url.replaceFirst("http://", "https://")
            else -> url
        }
    }

    Box(
        modifier = modifier
            .size(size)
            .clip(RoundedCornerShape(cornerRadius))
            .background(SpotifyCardHighlight),
        contentAlignment = Alignment.Center
    ) {
        if (!sanitizedUrl.isNullOrBlank()) {
            SubcomposeAsyncImage(
                model = ImageRequest.Builder(LocalContext.current)
                    .data(sanitizedUrl)
                    .crossfade(true)
                    .build(),
                contentDescription = "$stationName logo",
                contentScale = ContentScale.Crop,
                modifier = Modifier.fillMaxSize(),
                loading = {
                    StationLogoFallback(stationName = stationName, size = size)
                },
                error = {
                    StationLogoFallback(stationName = stationName, size = size)
                }
            )
        } else {
            StationLogoFallback(stationName = stationName, size = size)
        }
    }
}

@Composable
fun StationLogoFallback(
    stationName: String,
    size: Dp,
    modifier: Modifier = Modifier
) {
    val cleanName = stationName.trim()
    val hash = cleanName.hashCode()
    val gradients = listOf(
        listOf(Color(0xFF1DB954), Color(0xFF0D5325)), // Spotify Green
        listOf(Color(0xFF2E77D0), Color(0xFF103366)), // Electric Blue
        listOf(Color(0xFFE91429), Color(0xFF6B0913)), // Vibrant Red
        listOf(Color(0xFFE8630A), Color(0xFF752F02)), // Bright Amber Orange
        listOf(Color(0xFF7B2CBF), Color(0xFF3C096C)), // Purple
        listOf(Color(0xFF009688), Color(0xFF004D40)), // Deep Teal
        listOf(Color(0xFFFF007F), Color(0xFF800040))  // Hot Pink
    )
    val chosenGradient = gradients[(kotlin.math.abs(hash) % gradients.size)]

    Box(
        modifier = modifier
            .fillMaxSize()
            .background(Brush.linearGradient(chosenGradient)),
        contentAlignment = Alignment.Center
    ) {
        // Show 2-letter abbreviation for 3FM, 538, Qmusic, etc. or first initial
        val shortLabel = when {
            cleanName.startsWith("NPO", ignoreCase = true) -> cleanName.substringAfter("NPO").trim().take(3).ifBlank { "NPO" }
            cleanName.contains("538") -> "538"
            cleanName.contains("3FM", ignoreCase = true) -> "3FM"
            cleanName.contains("10", ignoreCase = true) -> "R10"
            cleanName.contains("JOE", ignoreCase = true) -> "JOE"
            cleanName.contains("SLAM", ignoreCase = true) -> "SLAM"
            else -> cleanName.take(2).uppercase()
        }

        if (size >= 36.dp && shortLabel.isNotBlank()) {
            Text(
                text = shortLabel,
                color = Color.White,
                fontSize = if (shortLabel.length > 2) (size.value * 0.28f).sp else (size.value * 0.38f).sp,
                fontWeight = FontWeight.Black,
                letterSpacing = (-0.5).sp
            )
        } else {
            Icon(
                imageVector = Icons.Default.Radio,
                contentDescription = null,
                tint = Color.White,
                modifier = Modifier.size(size * 0.5f)
            )
        }
    }
}

@Composable
fun WeatherCanvasIcon(
    condition: WeatherCondition,
    isDay: Boolean,
    modifier: Modifier = Modifier
) {
    Canvas(modifier = modifier) {
        val width = size.width
        val height = size.height

        when (condition) {
            WeatherCondition.SUNNY -> {
                drawSun(center = Offset(width / 2f, height / 2f), radius = width * 0.32f, color = Color(0xFFFFB300))
            }
            WeatherCondition.PARTLY_CLOUDY -> {
                // Sun slightly behind in top-right
                drawSun(center = Offset(width * 0.68f, height * 0.35f), radius = width * 0.22f, color = Color(0xFFFFB300))

            }
            WeatherCondition.CLOUDY -> {
                drawCloud(center = Offset(width * 0.5f, height * 0.52f), cloudWidth = width * 0.78f, cloudColor = Color(0xFFB0BEC5))
            }
            WeatherCondition.RAINY -> {
                // Cloud
                drawCloud(center = Offset(width * 0.5f, height * 0.42f), cloudWidth = width * 0.72f, cloudColor = Color(0xFF90A4AE))
                // Rain drops
                val dropColor = Color(0xFF4FC3F7)
                drawRainDrop(start = Offset(width * 0.35f, height * 0.68f), length = height * 0.22f, color = dropColor)
                drawRainDrop(start = Offset(width * 0.50f, height * 0.72f), length = height * 0.22f, color = dropColor)
                drawRainDrop(start = Offset(width * 0.65f, height * 0.68f), length = height * 0.22f, color = dropColor)
            }
            WeatherCondition.THUNDERSTORM -> {
                drawCloud(center = Offset(width * 0.5f, height * 0.42f), cloudWidth = width * 0.75f, cloudColor = Color(0xFF546E7A))
                // Lightning bolt
                val path = Path().apply {
                    moveTo(width * 0.52f, height * 0.55f)
                    lineTo(width * 0.42f, height * 0.75f)
                    lineTo(width * 0.52f, height * 0.75f)
                    lineTo(width * 0.45f, height * 0.95f)
                }
                drawPath(path = path, color = Color(0xFFFFD54F))
            }
            WeatherCondition.SNOWY -> {

private fun DrawScope.drawSun(center: Offset, radius: Float, color: Color) {
    drawCircle(color = color, radius = radius, center = center)
    // Subtle sun rays
    val rayLength = radius * 0.35f
    val rayColor = color.copy(alpha = 0.8f)
    for (i in 0 until 8) {
        val angle = (i * (kotlin.math.PI / 4)).toFloat()
        val startX = center.x + (radius + 2f) * kotlin.math.cos(angle)
        val startY = center.y + (radius + 2f) * kotlin.math.sin(angle)
        val endX = center.x + (radius + 2f + rayLength) * kotlin.math.cos(angle)
        val endY = center.y + (radius + 2f + rayLength) * kotlin.math.sin(angle)
        drawLine(
            color = rayColor,
            start = Offset(startX, startY),
            end = Offset(endX, endY),
            strokeWidth = 3f,
            cap = StrokeCap.Round
        )
    }
}

private fun DrawScope.drawCloud(center: Offset, cloudWidth: Float, cloudColor: Color) {
    val r = cloudWidth * 0.26f
    // Base fluffy circles
    drawCircle(color = cloudColor, radius = r * 0.8f, center = Offset(center.x - r * 0.9f, center.y + r * 0.1f))
    drawCircle(color = cloudColor, radius = r * 1.15f, center = Offset(center.x - r * 0.2f, center.y - r * 0.25f))
    drawCircle(color = cloudColor, radius = r * 0.85f, center = Offset(center.x + r * 0.8f, center.y + r * 0.15f))
    // Soft base flat pill
    drawRoundRect(
        color = cloudColor,
        topLeft = Offset(center.x - r * 1.2f, center.y),
        size = androidx.compose.ui.geometry.Size(r * 2.5f, r * 0.8f),
        cornerRadius = androidx.compose.ui.geometry.CornerRadius(r * 0.4f, r * 0.4f)
    )
}

private fun DrawScope.drawRainDrop(start: Offset, length: Float, color: Color) {
    drawLine(
        color = color,
        start = start,
        end = Offset(start.x - 3f, start.y + length),
        strokeWidth = 3f,
        cap = StrokeCap.Round
    )
}

/**
 * Domain model representing a global radio station.
 */
data class RadioStation(
    val stationUuid: String,
    val name: String,
    val streamUrl: String,
    val favicon: String = "",
    val homepage: String = "",
    val country: String = "",
    val countryCode: String = "",
    val tags: String = "",
    val codec: String = "MP3",
    val bitrate: Int = 128,
    val votes: Int = 0,
    val isFavorite: Boolean = false,
    val lastPlayedTimestamp: Long = 0L
) {
    val genreDisplay: String
        get() {
            if (tags.isBlank()) return "Diverse / Pop"
            return tags.split(",")
                .take(3)
                .joinToString(" • ") { it.trim().replaceFirstChar { char -> char.uppercase() } }
        }

    val displayBitrate: String
        get() = if (bitrate > 0) "$bitrate kbps" else "HQ Stream"

    val displayCountryFlag: String
        get() {
            val code = countryCode.trim().uppercase()
            if (code.length != 2) return "📻"
            val firstChar = Character.codePointAt(code, 0) - 0x41 + 0x1F1E6
            val secondChar = Character.codePointAt(code, 1) - 0x41 + 0x1F1E6
            return String(Character.toChars(firstChar)) + String(Character.toChars(secondChar))
        }
}

package com.example.model

enum class WeatherCondition {
    SUNNY,              // Zonnetje
    PARTLY_CLOUDY,      // Wolkje met zonnetje er iets achter
    CLOUDY,             // Bewolkt
    RAINY,              // Wolkje met regen
    THUNDERSTORM,       // Onweer
    SNOWY               // Sneeuw
}

data class WeatherInfo(
    val temperatureCelsius: Int,
    val condition: WeatherCondition,
    val conditionDescription: String,
    val isDay: Boolean = true
) {
    companion object {
        fun fromWmoCode(code: Int?, temp: Double?, isDayTime: Boolean = true): WeatherInfo {
            val roundedTemp = temp?.let { kotlin.math.round(it).toInt() } ?: 18
            val c = code ?: 0

            val condition = when (c) {
                0, 1 -> WeatherCondition.SUNNY
                2 -> WeatherCondition.PARTLY_CLOUDY
                3, 45, 48 -> WeatherCondition.CLOUDY
                51, 53, 55, 61, 63, 65, 80, 81, 82 -> WeatherCondition.RAINY
                95, 96, 99 -> WeatherCondition.THUNDERSTORM
                71, 73, 75, 77, 85, 86 -> WeatherCondition.SNOWY
                else -> WeatherCondition.PARTLY_CLOUDY
            }

            val desc = when (condition) {
                WeatherCondition.SUNNY -> if (isDayTime) "Zonnig" else "Helder"
                WeatherCondition.PARTLY_CLOUDY -> "Licht bewolkt"
                WeatherCondition.CLOUDY -> "Bewolkt"
                WeatherCondition.RAINY -> "Regen"
                WeatherCondition.THUNDERSTORM -> "Onweer"
                WeatherCondition.SNOWY -> "Sneeuw"
            }

            return WeatherInfo(
                temperatureCelsius = roundedTemp,
                condition = condition,
                conditionDescription = desc,
                isDay = isDayTime
            )
        }
    }
}

import okhttp3.OkHttpClient

import java.util.concurrent.TimeUnit

interface RadioBrowserApiService {

    @GET("json/stations/topclick")
    suspend fun getTopClickStations(
        @Query("limit") limit: Int = 40,
        @Query("hidebroken") hidebroken: Boolean = true
    ): List<StationDto>

    @GET("json/stations/topvote")
    suspend fun getTopVotedStations(
        @Query("limit") limit: Int = 40,
        @Query("hidebroken") hidebroken: Boolean = true
    ): List<StationDto>

    @GET("json/stations/bycountrycodeexact/{countryCode}")
    suspend fun getStationsByCountry(
        @Path("countryCode") countryCode: String,
        @Query("limit") limit: Int = 40,
        @Query("order") order: String = "votes",
        @Query("reverse") reverse: Boolean = true,
        @Query("hidebroken") hidebroken: Boolean = true
    ): List<StationDto>

    @GET("json/stations/bytag/{tag}")
    suspend fun getStationsByTag(
        @Path("tag") tag: String,
        @Query("limit") limit: Int = 40,
        @Query("order") order: String = "votes",
        @Query("reverse") reverse: Boolean = true,
        @Query("hidebroken") hidebroken: Boolean = true
    ): List<StationDto>

    @GET("json/stations/search")
    suspend fun searchStations(
        @Query("name") name: String? = null,
        @Query("countrycode") countryCode: String? = null,
        @Query("tag") tag: String? = null,
        @Query("limit") limit: Int = 50,
        @Query("order") order: String = "votes",
        @Query("reverse") reverse: Boolean = true,
        @Query("hidebroken") hidebroken: Boolean = true
    ): List<StationDto>

    companion object {
        private const val BASE_URL = "https://de1.api.radio-browser.info/"

        fun create(): RadioBrowserApiService {
            val logging = HttpLoggingInterceptor().apply {
                level = HttpLoggingInterceptor.Level.BASIC
            }

            val client = OkHttpClient.Builder()
                .addInterceptor { chain ->
                    val request = chain.request().newBuilder()
                        .header("User-Agent", "AutoRadioAndroid/1.0")
                        .build()
                    chain.proceed(request)
                }
                .addInterceptor(logging)
                .connectTimeout(15, TimeUnit.SECONDS)
                .readTimeout(15, TimeUnit.SECONDS)
                .build()

            return Retrofit.Builder()
                .baseUrl(BASE_URL)
                .client(client)
                .addConverterFactory(MoshiConverterFactory.create())
                .build()
                .create(RadioBrowserApiService::class.java)
        }
    }
}

package com.example.data.repository

import com.example.data.CuratedStations
import com.example.data.local.FavoriteStationDao
import com.example.data.local.FavoriteStationEntity
import com.example.data.remote.RadioBrowserApiService

import kotlinx.coroutines.Dispatchers

class RadioRepository(
    private val apiService: RadioBrowserApiService,
    private val favoriteDao: FavoriteStationDao
) {

    val favoriteStations: Flow<List<RadioStation>> = favoriteDao.getAllFavorites()
        .map { entities ->
            entities.map { entity ->
                RadioStation(
                    stationUuid = entity.stationUuid,
                    name = entity.name,
                    streamUrl = entity.streamUrl,
                    favicon = entity.favicon,
                    homepage = entity.homepage,
                    country = entity.country,
                    countryCode = entity.countryCode,
                    tags = entity.tags,
                    codec = entity.codec,
                    bitrate = entity.bitrate,
                    votes = entity.votes,
                    isFavorite = true,
                    lastPlayedTimestamp = entity.lastPlayedTimestamp
                )
            }
        }.flowOn(Dispatchers.IO)

    val recentStations: Flow<List<RadioStation>> = favoriteDao.getRecentHistory()
        .map { entities ->
            entities.map { entity ->
                RadioStation(
                    stationUuid = entity.stationUuid,
                    name = entity.name,
                    streamUrl = entity.streamUrl,
                    favicon = entity.favicon,
                    homepage = entity.homepage,
                    country = entity.country,
                    countryCode = entity.countryCode,
                    tags = entity.tags,
                    codec = entity.codec,
                    bitrate = entity.bitrate,
                    votes = entity.votes,
                    isFavorite = entity.isFavorite,
                    lastPlayedTimestamp = entity.lastPlayedTimestamp
                )
            }
        }.flowOn(Dispatchers.IO)

    suspend fun isFavorite(uuid: String): Boolean = withContext(Dispatchers.IO) {
        val entity = favoriteDao.getFavoriteById(uuid)
        entity?.isFavorite == true
    }

    suspend fun toggleFavorite(station: RadioStation): Boolean = withContext(Dispatchers.IO) {
        val existing = favoriteDao.getFavoriteById(station.stationUuid)
        if (existing != null) {
            val newFav = !existing.isFavorite
            favoriteDao.insertOrUpdate(
                existing.copy(
                    isFavorite = newFav,
                    addedTimestamp = if (newFav) System.currentTimeMillis() else 0L
                )
            )
            newFav
        } else {
            favoriteDao.insertOrUpdate(
                FavoriteStationEntity(
                    stationUuid = station.stationUuid,
                    name = station.name,
                    streamUrl = station.streamUrl,
                    favicon = station.favicon,
                    homepage = station.homepage,
                    country = station.country,
                    countryCode = station.countryCode,
                    tags = station.tags,
                    codec = station.codec,
                    bitrate = station.bitrate,
                    votes = station.votes,
                    isFavorite = true,
                    addedTimestamp = System.currentTimeMillis()
                )
            )
            true
        }
    }

    suspend fun recordPlayed(station: RadioStation) = withContext(Dispatchers.IO) {
        val existing = favoriteDao.getFavoriteById(station.stationUuid)
        val now = System.currentTimeMillis()
        if (existing != null) {
            favoriteDao.updateLastPlayed(station.stationUuid, now)
        } else {
            // CRITICAL: isFavorite MUST be false so it only logs play history and NEVER becomes a favorite!
            favoriteDao.insertOrUpdate(
                FavoriteStationEntity(
                    stationUuid = station.stationUuid,
                    name = station.name,
                    streamUrl = station.streamUrl,
                    favicon = station.favicon,
                    homepage = station.homepage,
                    country = station.country,
                    countryCode = station.countryCode,
                    tags = station.tags,
                    codec = station.codec,
                    bitrate = station.bitrate,
                    votes = station.votes,
                    isFavorite = false,
                    addedTimestamp = 0L,
                    lastPlayedTimestamp = now
                )
            )
        }
    }

    suspend fun getTopStations(): List<RadioStation> = withContext(Dispatchers.IO) {
        val favIds = favoriteDao.getAllFavorites().first().map { it.stationUuid }.toSet()
        val curated = CuratedStations.defaultStations.map { it.copy(isFavorite = favIds.contains(it.stationUuid)) }
        try {
            val dtoList = apiService.getTopVotedStations(limit = 40)
            val remote = dtoList.mapNotNull { it.toDomain(favIds) }
            val merged = (curated + remote).distinctBy { it.name.trim().lowercase() }
            merged
        } catch (_: Exception) {
            curated
        }
    }

    suspend fun getDutchStations(): List<RadioStation> = withContext(Dispatchers.IO) {
        val favIds = favoriteDao.getAllFavorites().first().map { it.stationUuid }.toSet()
        val curatedDutch = CuratedStations.defaultStations
            .filter { it.countryCode == "NL" }
            .map { it.copy(isFavorite = favIds.contains(it.stationUuid)) }
        try {
            val dtoList = apiService.getStationsByCountry(countryCode = "NL", limit = 40)
            val remote = dtoList.mapNotNull { it.toDomain(favIds) }
            val merged = (curatedDutch + remote).distinctBy { it.name.trim().lowercase() }
            merged
        } catch (_: Exception) {
            curatedDutch
        }
    }

    suspend fun getStationsByCountry(countryCode: String): List<RadioStation> = withContext(Dispatchers.IO) {
        val favIds = favoriteDao.getAllFavorites().first().map { it.stationUuid }.toSet()
        val curatedMatching = CuratedStations.defaultStations
            .filter { it.countryCode.equals(countryCode, ignoreCase = true) }
            .map { it.copy(isFavorite = favIds.contains(it.stationUuid)) }
        try {
            val dtoList = apiService.getStationsByCountry(countryCode = countryCode, limit = 40)
            val remote = dtoList.mapNotNull { it.toDomain(favIds) }
            val merged = (curatedMatching + remote).distinctBy { it.name.trim().lowercase() }
            if (merged.isNotEmpty()) merged else curatedMatching
        } catch (_: Exception) {
            curatedMatching
        }
    }

    suspend fun getStationsByGenre(genre: String): List<RadioStation> = withContext(Dispatchers.IO) {
        val favIds = favoriteDao.getAllFavorites().first().map { it.stationUuid }.toSet()
        val curatedGenre = CuratedStations.defaultStations
            .filter { it.tags.contains(genre, ignoreCase = true) }
            .map { it.copy(isFavorite = favIds.contains(it.stationUuid)) }
        try {
            val dtoList = apiService.getStationsByTag(tag = genre.lowercase(), limit = 40)
            val remote = dtoList.mapNotNull { it.toDomain(favIds) }
            val merged = (curatedGenre + remote).distinctBy { it.name.trim().lowercase() }
            if (merged.isNotEmpty()) merged else curatedGenre
        } catch (_: Exception) {
            curatedGenre
        }
    }

    suspend fun searchStations(query: String, countryCode: String? = null, genre: String? = null): List<RadioStation> =
        withContext(Dispatchers.IO) {
            val favIds = favoriteDao.getAllFavorites().first().map { it.stationUuid }.toSet()
            val curatedMatches = CuratedStations.defaultStations.filter {
                (query.isBlank() || it.name.contains(query, ignoreCase = true) || it.tags.contains(query, ignoreCase = true)) &&
                        (countryCode.isNullOrBlank() || it.countryCode.equals(countryCode, ignoreCase = true)) &&
                        (genre.isNullOrBlank() || it.tags.contains(genre, ignoreCase = true))
            }.map { it.copy(isFavorite = favIds.contains(it.stationUuid)) }

            try {
                val dtoList = apiService.searchStations(
                    name = query.ifBlank { null },
                    countryCode = countryCode?.ifBlank { null },
                    tag = genre?.ifBlank { null },
                    limit = 50
                )
                val remote = dtoList.mapNotNull { it.toDomain(favIds) }
                (curatedMatches + remote).distinctBy { it.name.trim().lowercase() }
            } catch (_: Exception) {
                curatedMatches
            }
        }

    private fun StationDto.toDomain(favoriteIds: Set<String>): RadioStation? {
        val uuid = stationuuid ?: return null
        val stream = urlResolved?.takeIf { it.isNotBlank() } ?: url?.takeIf { it.isNotBlank() } ?: return null
        
        // Filter out stations that fail recent health checks or have invalid stream schemes
        if (lastcheckok != null && lastcheckok == 0) return null
        if (!stream.startsWith("http://") && !stream.startsWith("https://")) return null

        val stationName = name?.trim()?.takeIf { it.isNotBlank() } ?: "Radiozender"
        return RadioStation(
            stationUuid = uuid,
            name = stationName,
            streamUrl = stream,
            favicon = favicon?.trim() ?: "",
            homepage = homepage?.trim() ?: "",
            country = country?.trim() ?: "",
            countryCode = countrycode?.trim() ?: "",
            tags = tags?.trim() ?: "",
            codec = codec?.trim() ?: "MP3",
            bitrate = bitrate ?: 128,
            votes = votes ?: 0,
            isFavorite = favoriteIds.contains(uuid)
        )
    }
}

@Entity(tableName = "favorite_stations")
data class FavoriteStationEntity(
    @PrimaryKey
    val stationUuid: String,
    val name: String,
    val streamUrl: String,
    val favicon: String,
    val homepage: String,
    val country: String,
    val countryCode: String,
    val tags: String,
    val codec: String,
    val bitrate: Int,
    val votes: Int,
    val isFavorite: Boolean = false,
    val addedTimestamp: Long = 0L,
    val lastPlayedTimestamp: Long = 0L
)

@Dao
interface FavoriteStationDao {
    @Query("SELECT * FROM favorite_stations WHERE isFavorite = 1 ORDER BY addedTimestamp DESC")
    fun getAllFavorites(): Flow<List<FavoriteStationEntity>>

    @Query("SELECT * FROM favorite_stations WHERE stationUuid = :uuid LIMIT 1")
    suspend fun getFavoriteById(uuid: String): FavoriteStationEntity?

    @Query("SELECT EXISTS(SELECT 1 FROM favorite_stations WHERE stationUuid = :uuid AND isFavorite = 1)")
    fun isFavorite(uuid: String): Flow<Boolean>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertOrUpdate(station: FavoriteStationEntity)

    @Query("UPDATE favorite_stations SET isFavorite = 0, addedTimestamp = 0 WHERE stationUuid = :uuid")
    suspend fun removeFavorite(uuid: String)

    @Query("SELECT * FROM favorite_stations WHERE lastPlayedTimestamp > 0 ORDER BY lastPlayedTimestamp DESC LIMIT 15")
    fun getRecentHistory(): Flow<List<FavoriteStationEntity>>

    @Query("UPDATE favorite_stations SET lastPlayedTimestamp = :timestamp WHERE stationUuid = :uuid")
    suspend fun updateLastPlayed(uuid: String, timestamp: Long)
}

                description = "Toont bediening voor de radio op de achtergrond en Android Auto"
                setShowBadge(false)
            }
            val manager = getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager
            manager?.createNotificationChannel(channel)
        }
    }

    private fun initializeSession() {
        if (mediaLibrarySession != null) return

        val intent = Intent(this, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP
        }
        val pendingIntent = PendingIntent.getActivity(
            this,
            0,
            intent,
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )

        mediaLibrarySession = MediaLibrarySession.Builder(this, playerManager.exoPlayer, MediaLibraryCallback())
            .setSessionActivity(pendingIntent)
            .setId("AutoRadioSession_${System.currentTimeMillis()}")
            .build()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        initializeSession()
        super.onStartCommand(intent, flags, startId)
        // START_STICKY ensures Android OS restarts the service automatically if memory pressure drops it
        return START_STICKY
    }

    override fun onGetSession(controllerInfo: MediaSession.ControllerInfo): MediaLibrarySession? {

                        }

                            val favorites = withTimeoutOrNull(1500) { repository.favoriteStations.first() }
                            val items = if (!favorites.isNullOrEmpty()) {
                                favorites.map { it.toMediaItem() }
                            } else {
                                // If no favorites saved yet, show curated top stations as helpful suggestions
                                CuratedStations.defaultStations.take(6).map { it.toMediaItem() }
                            }
                            future.set(LibraryResult.ofItemList(ImmutableList.copyOf(items), params))
                        }

                        CATEGORY_TOP -> {
                            val stations = withTimeoutOrNull(2000) { repository.getTopStations() }
                                ?: CuratedStations.defaultStations
                            val items = stations.map { it.toMediaItem() }
                            future.set(LibraryResult.ofItemList(ImmutableList.copyOf(items), params))
                        }

                        CATEGORY_GENRES -> {
                            val genres = listOf(
                                Pair("Dance", "Club & Electronic"),
                                Pair("Rock", "Classic & Modern Rock"),
                                Pair("Pop", "Top 40 & Hits"),
                                Pair("News", "Nieuws & Actualiteiten"),
                                Pair("Jazz", "Smooth Jazz & Blues"),
                                Pair("Classical", "Klassieke Muziek")
                            )
                            val genreItems = genres.map { (genre, subtitle) ->
                                createFolderItem("genre_$genre", genre, subtitle)
                            }
                            future.set(LibraryResult.ofItemList(ImmutableList.copyOf(genreItems), params))
                        }

                        else -> {
                            if (parentId.startsWith("genre_")) {
                                val tag = parentId.removePrefix("genre_")
                                val stations = withTimeoutOrNull(2000) { repository.getStationsByGenre(tag) }
                                    ?: CuratedStations.defaultStations.filter { it.tags.contains(tag, ignoreCase = true) }
                                val items = stations.map { it.toMediaItem() }
                                future.set(LibraryResult.ofItemList(ImmutableList.copyOf(items), params))
                            } else {
                                future.set(LibraryResult.ofItemList(ImmutableList.of(), params))
                            }
                        }
                    }
                } catch (e: Exception) {
                    // Safe instant fallback to curated stations so Android Auto NEVER shows "geen inhoud kan laden"
                    val fallback = CuratedStations.defaultStations.map { it.toMediaItem() }
                    future.set(LibraryResult.ofItemList(ImmutableList.copyOf(fallback), params))
                }
            }

            return future
        }

        // Handle item selection from Android Auto UI
        override fun onSetMediaItems(
            mediaSession: MediaSession,
            controller: MediaSession.ControllerInfo,
            mediaItems: MutableList<MediaItem>,
            startIndex: Int,
            startPositionMs: Long
        ): ListenableFuture<MediaSession.MediaItemsWithStartPosition> {
            val resolvedItems = mediaItems.map { item ->
                if (item.localConfiguration?.uri != null) {
                    item
                } else {
                    val station = CuratedStations.defaultStations.find { it.stationUuid == item.mediaId }
                    if (station != null) {
                        station.toMediaItem()
                    } else {
                        item
                    }
                }
            }

            // Also synchronize with playerManager's currentStation state
            resolvedItems.getOrNull(startIndex)?.let { chosen ->
                val matchedStation = CuratedStations.defaultStations.find { it.stationUuid == chosen.mediaId }
                    ?: RadioStation(
                        stationUuid = chosen.mediaId,
                        name = chosen.mediaMetadata.title?.toString() ?: "Radio",
                        streamUrl = chosen.localConfiguration?.uri?.toString() ?: "",
                        favicon = chosen.mediaMetadata.artworkUri?.toString() ?: "",
                        country = chosen.mediaMetadata.albumTitle?.toString() ?: "NL"
                    )
                serviceScope.launch {
                    playerManager.playStation(matchedStation)
                }
            }

            return Futures.immediateFuture(
                MediaSession.MediaItemsWithStartPosition(resolvedItems, startIndex, startPositionMs)
            )
        }

        override fun onCustomCommand(
            session: MediaSession,
            controller: MediaSession.ControllerInfo,
            customCommand: SessionCommand,
            args: Bundle
        ): ListenableFuture<SessionResult> {
            return Futures.immediateFuture(SessionResult(SessionResult.RESULT_SUCCESS))
        }

        private fun createFolderItem(id: String, title: String, subtitle: String): MediaItem {
            return MediaItem.Builder()
                .setMediaId(id)
                .setMediaMetadata(
                    MediaMetadata.Builder()
                        .setTitle(title)
                        .setSubtitle(subtitle)
                        .setIsBrowsable(true)
                        .setIsPlayable(false)
                        .setMediaType(MediaMetadata.MEDIA_TYPE_FOLDER_MIXED)
                        .build()
                )
                .build()
        }

        private fun RadioStation.toMediaItem(): MediaItem {
            // Generate or fetch local content:// uri so Android Auto displays icons with 100% reliability
            val iconContentUri = StationIconProvider.getIconUri(
                context = this@RadioMediaService,
                stationUuid = stationUuid,
                stationName = name,
                remoteUrl = favicon
            )

            return MediaItem.Builder()
                .setMediaId(stationUuid)
                .setUri(streamUrl)
                .setMediaMetadata(
                    MediaMetadata.Builder()
                        .setTitle(name)
                        .setArtist(genreDisplay)
                        .setAlbumTitle(country)
                        .setArtworkUri(iconContentUri)
                        .setIsBrowsable(false)
                        .setIsPlayable(true)
                        .setMediaType(MediaMetadata.MEDIA_TYPE_RADIO_STATION)
                        .build()
                )
                .build()
        }
    }

    override fun onDestroy() {
        serviceScope.cancel()
        mediaLibrarySession?.run {
            // Note: Do NOT release the shared playerManager.exoPlayer here, as that would invalidate
            // the singleton player if Android Auto disconnects and reconnects while app is still in memory.
            release()
            mediaLibrarySession = null
        }
        super.onDestroy()
    }
}

/**
 * Serves locally cached station icons to Android Auto and system UI via content:// URIs.
 * This guarantees Android Auto can load artwork without remote network security blocks.
 */
class StationIconProvider : ContentProvider() {

    companion object {
        const val AUTHORITY = "com.aistudio.autoradio.kxrv.icons"
        val CONTENT_URI: Uri = Uri.parse("content://$AUTHORITY")

        private val cacheScope = CoroutineScope(Dispatchers.IO + SupervisorJob())

        fun getIconUri(context: Context, stationUuid: String, stationName: String, remoteUrl: String?): Uri {
            val key = generateKey(stationUuid)
            val file = File(context.cacheDir, "auto_icons/$key.png")

            // If not cached yet, trigger background download or generate fallback bitmap
            if (!file.exists()) {
                cacheScope.launch {
                    cacheIcon(context, key, stationName, remoteUrl)
                }
            }

            return Uri.parse("content://$AUTHORITY/$key.png")
        }

        fun preloadIcon(context: Context, stationUuid: String, stationName: String, remoteUrl: String?) {
            val key = generateKey(stationUuid)
            val file = File(context.cacheDir, "auto_icons/$key.png")
            if (!file.exists()) {
                cacheScope.launch {
                    cacheIcon(context, key, stationName, remoteUrl)
                }
            }
        }

        private suspend fun cacheIcon(context: Context, key: String, stationName: String, remoteUrl: String?) {
            withContext(Dispatchers.IO) {
                try {
                    val dir = File(context.cacheDir, "auto_icons")
                    if (!dir.exists()) dir.mkdirs()
                    val targetFile = File(dir, "$key.png")
                    if (targetFile.exists() && targetFile.length() > 0) return@withContext

                    var bitmap: Bitmap? = null

                    // Attempt downloading via Coil if URL is valid
                    if (!remoteUrl.isNullOrBlank()) {
                        val sanitized = when {
                            remoteUrl.contains("upload.wikimedia.org") && remoteUrl.contains("/thumb/") -> null
                            remoteUrl.startsWith("http://") -> remoteUrl.replaceFirst("http://", "https://")
                            else -> remoteUrl
                        }

                        if (!sanitized.isNullOrBlank()) {
                            try {
                                val imageLoader = coil.Coil.imageLoader(context)
                                val request = ImageRequest.Builder(context)
                                    .data(sanitized)
                                    .allowHardware(false)
                                    .size(256, 256)
                                    .build()
                                val result = imageLoader.execute(request)
                                if (result is SuccessResult) {
                                    val drawable = result.drawable
                                    bitmap = Bitmap.createBitmap(
                                        256, 256,
                                        Bitmap.Config.ARGB_8888
                                    )
                                    val canvas = Canvas(bitmap)
                                    drawable.setBounds(0, 0, 256, 256)
                                    drawable.draw(canvas)
                                }
                            } catch (_: Exception) {
                                bitmap = null
                            }
                        }
                    }

                    // Fallback to high quality generated badge
                    if (bitmap == null) {
                        bitmap = generateFallbackBitmap(stationName)
                    }

                    // Save to local cache file
                    FileOutputStream(targetFile).use { out: OutputStream ->
                        bitmap.compress(Bitmap.CompressFormat.PNG, 95, out)
                    }
                } catch (_: Exception) {
                }
            }
        }

        private fun generateFallbackBitmap(stationName: String): Bitmap {
            val size = 256
            val bitmap = Bitmap.createBitmap(size, size, Bitmap.Config.ARGB_8888)
            val canvas = Canvas(bitmap)

            val cleanName = stationName.trim()
            val hash = kotlin.math.abs(cleanName.hashCode())

            val colors = listOf(
                Pair(0xFF1DB954.toInt(), 0xFF0D5325.toInt()), // Green
                Pair(0xFF2E77D0.toInt(), 0xFF103366.toInt()), // Blue
                Pair(0xFFE91429.toInt(), 0xFF6B0913.toInt()), // Red
                Pair(0xFFE8630A.toInt(), 0xFF752F02.toInt()), // Orange
                Pair(0xFF7B2CBF.toInt(), 0xFF3C096C.toInt()), // Purple
                Pair(0xFF009688.toInt(), 0xFF004D40.toInt())  // Teal
            )
            val (c1, c2) = colors[hash % colors.size]

            val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
                color = c1
            }
            canvas.drawRoundRect(RectF(0f, 0f, size.toFloat(), size.toFloat()), 32f, 32f, paint)

            val shortLabel = when {
                cleanName.startsWith("NPO", ignoreCase = true) -> cleanName.substringAfter("NPO").trim().take(3).ifBlank { "NPO" }
                cleanName.contains("538") -> "538"
                cleanName.contains("3FM", ignoreCase = true) -> "3FM"
                cleanName.contains("10", ignoreCase = true) -> "R10"
                cleanName.contains("JOE", ignoreCase = true) -> "JOE"
                cleanName.contains("SLAM", ignoreCase = true) -> "SLAM"
                else -> cleanName.take(2).uppercase()
            }

            val textPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
                color = android.graphics.Color.WHITE
                textSize = if (shortLabel.length > 2) 72f else 96f
                isFakeBoldText = true
                textAlign = Paint.Align.CENTER
            }

            val textY = (size / 2f) - ((textPaint.descent() + textPaint.ascent()) / 2f)
            canvas.drawText(shortLabel, size / 2f, textY, textPaint)

            return bitmap
        }

        private fun generateKey(input: String): String {
            return try {
                val md = MessageDigest.getInstance("MD5")
                val bytes = md.digest(input.toByteArray())
                bytes.joinToString("") { "%02x".format(it) }
            } catch (_: Exception) {
                input.replace(Regex("[^a-zA-Z0-9]"), "_")
            }
        }
    }

    override fun onCreate(): Boolean = true

    override fun query(uri: Uri, projection: Array<out String>?, selection: String?, selectionArgs: Array<out String>?, sortOrder: String?): Cursor? = null

    override fun getType(uri: Uri): String = "image/png"

    override fun insert(uri: Uri, values: ContentValues?): Uri? = null

    override fun delete(uri: Uri, selection: String?, selectionArgs: Array<out String>?): Int = 0

    override fun update(uri: Uri, values: ContentValues?, selection: String?, selectionArgs: Array<out String>?): Int = 0

    override fun openFile(uri: Uri, mode: String): ParcelFileDescriptor? {
        val fileName = uri.lastPathSegment ?: return null
        val context = context ?: return null
        val file = File(context.cacheDir, "auto_icons/$fileName")

        // If file doesn't exist yet, generate standard placeholder immediately so Android Auto never hangs
        if (!file.exists()) {
            val dir = File(context.cacheDir, "auto_icons")
            if (!dir.exists()) dir.mkdirs()
            val placeholder = generateFallbackBitmap("Radio")
            FileOutputStream(file).use { out ->
                placeholder.compress(Bitmap.CompressFormat.PNG, 95, out)
            }
        }

        return ParcelFileDescriptor.open(file, ParcelFileDescriptor.MODE_READ_ONLY)
    }
}

@OptIn(UnstableApi::class)
class RadioPlayerManager(context: Context) {

    private val appContext = context.applicationContext
    private val scope = CoroutineScope(Dispatchers.Main + Job())

    // High compatibility HTTP DataSource: supports redirects, Icecast, Shoutcast, HLS & standard ICY headers
    private val httpDataSourceFactory = DefaultHttpDataSource.Factory()
        .setUserAgent("AutoRadio/1.0 (Android; Linux; ExoPlayer)")
        .setAllowCrossProtocolRedirects(true)
        .setConnectTimeoutMs(15000)
        .setReadTimeoutMs(20000)
        .setKeepPostFor302Redirects(true)

    // Extractors configured for internet radio formats (MP3, AAC, OGG/Opus, TS/HLS, FLAC)
    private val extractorsFactory = DefaultExtractorsFactory()
        .setConstantBitrateSeekingEnabled(true)
        .setTsExtractorFlags(DefaultTsPayloadReaderFactory.FLAG_ALLOW_NON_IDR_KEYFRAMES)

    private val mediaSourceFactory = DefaultMediaSourceFactory(httpDataSourceFactory, extractorsFactory)

    // Optimized buffering for live mobile / car radio (quick initial start + generous max buffer)
    private val loadControl = DefaultLoadControl.Builder()
        .setBufferDurationsMs(
            /* minBufferMs = */ 3000,
            /* maxBufferMs = */ 30000,
            /* bufferForPlaybackMs = */ 1500,
            /* bufferForPlaybackAfterRebufferMs = */ 3000
        )
        .build()

    val exoPlayer: ExoPlayer = ExoPlayer.Builder(
        appContext,
        DefaultRenderersFactory(appContext).setExtensionRendererMode(DefaultRenderersFactory.EXTENSION_RENDERER_MODE_PREFER)
    )
        .setMediaSourceFactory(mediaSourceFactory)
        .setLoadControl(loadControl)
        .setAudioAttributes(

                .setContentType(C.AUDIO_CONTENT_TYPE_MUSIC)
                .setUsage(C.USAGE_MEDIA)
                .build(),
            /* handleAudioFocus = */ true
        )
        .setHandleAudioBecomingNoisy(true)
        .setWakeMode(C.WAKE_MODE_NETWORK)
        .build()

    private val _currentStation = MutableStateFlow<RadioStation?>(null)
    val currentStation: StateFlow<RadioStation?> = _currentStation.asStateFlow()

    private val _playbackStatus = MutableStateFlow(PlaybackStatus.IDLE)
    val playbackStatus: StateFlow<PlaybackStatus> = _playbackStatus.asStateFlow()

    private val _rdsInfo = MutableStateFlow(RdsInfo())
    val rdsInfo: StateFlow<RdsInfo> = _rdsInfo.asStateFlow()

    private val _errorMessage = MutableStateFlow<String?>(null)
    val errorMessage: StateFlow<String?> = _errorMessage.asStateFlow()

    val playlist: StateFlow<List<RadioStation>> = _playlist.asStateFlow()

    val sleepTimerSeconds: StateFlow<Int?> = _sleepTimerSeconds.asStateFlow()
    private var sleepTimerJob: Job? = null

    private val _isMuted = MutableStateFlow(false)
    val isMuted: StateFlow<Boolean> = _isMuted.asStateFlow()

    private var previousVolume: Float = 1.0f
    private var retryAttempt = 0

    init {
        exoPlayer.addListener(object : Player.Listener {
            override fun onPlaybackStateChanged(state: Int) {
                when (state) {
                    Player.STATE_BUFFERING -> _playbackStatus.value = PlaybackStatus.BUFFERING
                    Player.STATE_READY -> {
                        retryAttempt = 0
                        _playbackStatus.value = if (exoPlayer.playWhenReady) {
                            PlaybackStatus.PLAYING
                        } else {
                            PlaybackStatus.PAUSED
                        }
                    }
                    Player.STATE_ENDED -> _playbackStatus.value = PlaybackStatus.IDLE
                    Player.STATE_IDLE -> {
                        if (_playbackStatus.value != PlaybackStatus.ERROR) {
                            _playbackStatus.value = PlaybackStatus.IDLE
                        }
                    }
                }
            }

            override fun onIsPlayingChanged(isPlaying: Boolean) {
                if (isPlaying) {
                    _playbackStatus.value = PlaybackStatus.PLAYING
                    _errorMessage.value = null
                } else if (_playbackStatus.value != PlaybackStatus.BUFFERING && _playbackStatus.value != PlaybackStatus.ERROR) {
                    _playbackStatus.value = PlaybackStatus.PAUSED
                }
            }

            override fun onPlayerError(error: PlaybackException) {
                val current = _currentStation.value
                if (current != null && retryAttempt < 1) {
                    retryAttempt++
                    scope.launch {
                        delay(1200)
                        // If it failed on https or http redirect, retry with alternate scheme if applicable
                        val altUrl = when {
                            current.streamUrl.startsWith("https://") -> current.streamUrl.replaceFirst("https://", "http://")
                            current.streamUrl.startsWith("http://") -> current.streamUrl.replaceFirst("http://", "https://")
                            else -> current.streamUrl
                        }
                        tryPlayStream(altUrl, current)
                    }
                } else {
                    _playbackStatus.value = PlaybackStatus.ERROR
                    _errorMessage.value = "Kon live audio stream niet laden. Probeer een andere zender."
                }
            }

            override fun onMediaMetadataChanged(mediaMetadata: MediaMetadata) {
                parseMediaMetadata(mediaMetadata)
            }

            override fun onMetadata(metadata: Metadata) {
                parseIcyMetadata(metadata)
            }

    fun playStation(station: RadioStation, newPlaylist: List<RadioStation>? = null) {
        if (newPlaylist != null) {
            _playlist.value = newPlaylist
        } else if (!_playlist.value.any { it.stationUuid == station.stationUuid }) {
            _playlist.value = _playlist.value + station
        }

        _currentStation.value = station
        _errorMessage.value = null
        _playbackStatus.value = PlaybackStatus.BUFFERING
        retryAttempt = 0

        // Initialize base RDS info from station
        _rdsInfo.value = RdsInfo(
            stationName = station.name,
            genre = station.genreDisplay,
            bitrate = station.bitrate,
            codec = station.codec,
            isLive = true
        )

        tryPlayStream(station.streamUrl, station)
    }

    private fun tryPlayStream(streamUrl: String, station: RadioStation) {
        try {
            val mediaMetadata = MediaMetadata.Builder()
                .setTitle(station.name)
                .setArtist("Live Radio • ${station.country}")
                .setArtworkUri(if (station.favicon.isNotBlank()) Uri.parse(station.favicon) else null)
                .build()

            val mediaItem = MediaItem.Builder()
                .setMediaId(station.stationUuid)
                .setUri(streamUrl)
                .setMediaMetadata(mediaMetadata)
                .build()

            exoPlayer.stop()
            exoPlayer.clearMediaItems()
            exoPlayer.setMediaItem(mediaItem)
            exoPlayer.prepare()
            exoPlayer.play()
        } catch (e: Exception) {
            _playbackStatus.value = PlaybackStatus.ERROR
            _errorMessage.value = "Fout bij starten van stream: ${e.localizedMessage}"
        }
    }

    fun updateCurrentStationFavorite(isFavorite: Boolean) {
        val current = _currentStation.value ?: return
        _currentStation.value = current.copy(isFavorite = isFavorite)
        _playlist.value = _playlist.value.map {
            if (it.stationUuid == current.stationUuid) it.copy(isFavorite = isFavorite) else it
        }
    }

    fun togglePlayPause() {
        if (_playbackStatus.value == PlaybackStatus.PLAYING) {
            exoPlayer.pause()
            _playbackStatus.value = PlaybackStatus.PAUSED
        } else if (_playbackStatus.value == PlaybackStatus.PAUSED) {
            exoPlayer.play()
            _playbackStatus.value = PlaybackStatus.PLAYING
        } else {
            _currentStation.value?.let { playStation(it) }
        }
    }

    fun nextStation() {
        val list = _playlist.value
        if (list.isEmpty()) return
        val current = _currentStation.value ?: return
        val currentIndex = list.indexOfFirst { it.stationUuid == current.stationUuid }
        val nextIndex = if (currentIndex != -1 && currentIndex < list.size - 1) {
            currentIndex + 1
        } else {
            0
        }
        playStation(list[nextIndex])
    }

    fun previousStation() {
        val list = _playlist.value
        if (list.isEmpty()) return
        val current = _currentStation.value ?: return
        val currentIndex = list.indexOfFirst { it.stationUuid == current.stationUuid }
        val prevIndex = if (currentIndex > 0) {
            currentIndex - 1
        } else {
            list.size - 1
        }
        playStation(list[prevIndex])
    }

    fun toggleMute() {
        if (_isMuted.value) {
            exoPlayer.volume = previousVolume.coerceAtLeast(0.2f)
            _isMuted.value = false
        } else {
            previousVolume = exoPlayer.volume
            exoPlayer.volume = 0f
            _isMuted.value = true
        }
    }

    fun setSleepTimer(minutes: Int?) {
        sleepTimerJob?.cancel()
        if (minutes == null || minutes <= 0) {
            _sleepTimerSeconds.value = null
            return
        }

        val totalSeconds = minutes * 60
        _sleepTimerSeconds.value = totalSeconds

        sleepTimerJob = scope.launch {
            var remaining = totalSeconds
            while (isActive && remaining > 0) {
                delay(1000)
                remaining--
                _sleepTimerSeconds.value = remaining
            }
            if (isActive) {
                exoPlayer.pause()
                _playbackStatus.value = PlaybackStatus.PAUSED
                _sleepTimerSeconds.value = null
            }
        }
    }

    private fun parseMediaMetadata(mediaMetadata: MediaMetadata) {
        val title = mediaMetadata.title?.toString()?.trim() ?: ""
        val artist = mediaMetadata.artist?.toString()?.trim() ?: ""

        if (title.isNotBlank() || artist.isNotBlank()) {
            val current = _rdsInfo.value
            val full = if (artist.isNotBlank() && title.isNotBlank()) "$artist - $title" else title.ifBlank { artist }
            _rdsInfo.value = current.copy(
                trackTitle = title,
                artist = artist,
                fullTitle = full
            )
            updateLiveRdsMediaMetadata(full, artist, title)
        }
    }

    private fun parseIcyMetadata(metadata: Metadata) {
        for (i in 0 until metadata.length()) {
            when (val entry = metadata.get(i)) {
                is IcyInfo -> {
                    val streamTitle = entry.title?.trim() ?: ""
                    if (streamTitle.isNotBlank()) {
                        val parts = streamTitle.split(" - ", limit = 2)
                        val artist = if (parts.size > 1) parts[0].trim() else ""
                        val title = if (parts.size > 1) parts[1].trim() else parts[0].trim()

                        val current = _rdsInfo.value
                        _rdsInfo.value = current.copy(
                            fullTitle = streamTitle,
                            artist = artist,
                            trackTitle = title,
                            isLive = true
                        )
                        updateLiveRdsMediaMetadata(streamTitle, artist, title)
                    }
                }
                is IcyHeaders -> {
                    val current = _rdsInfo.value
                    _rdsInfo.value = current.copy(
                        bitrate = if (entry.bitrate > 0) entry.bitrate else current.bitrate,
                        genre = entry.genre?.trim()?.ifBlank { current.genre } ?: current.genre,
                        stationName = entry.name?.trim()?.ifBlank { current.stationName } ?: current.stationName
                    )
                }
            }
        }
    }

    private fun updateLiveRdsMediaMetadata(streamTitle: String, artist: String, trackTitle: String) {
        val current = _currentStation.value ?: return
        try {
            // Note: In ExoPlayer for live streams, calling replaceMediaItem during playback triggers 
            // a full stream re-buffer / stall. We update the UI StateFlow (_rdsInfo) directly, which
            // immediately reflects in Compose UI and MediaNotification without interrupting audio.
        } catch (_: Exception) {
        }
    }

    fun release() {
        sleepTimerJob?.cancel()
        exoPlayer.release()
    }

    companion object {
        @Volatile
        private var INSTANCE: RadioPlayerManager? = null

        fun getInstance(context: Context): RadioPlayerManager {
            return INSTANCE ?: synchronized(this) {
                val instance = RadioPlayerManager(context.applicationContext)
                INSTANCE = instance
                instance
            }
        }
    }
}
